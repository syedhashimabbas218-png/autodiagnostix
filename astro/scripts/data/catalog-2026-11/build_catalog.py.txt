def save_image(src, slug, index):
    os.makedirs(f'{OUT}/images/{slug}', exist_ok=True)
    rel = f'images/{slug}/{index}.webp'
    im = Image.open(src)
    if im.mode in ('P', 'LA', 'RGBA'):
        bg = Image.new('RGB', im.size, 'white')
        im = im.convert('RGBA')
        bg.paste(im, mask=im.split()[3])
        im = bg
    else:
        im = im.convert('RGB')
    im.thumbnail((1200, 1200))
    im.save(f'{OUT}/{rel}', 'WEBP', quality=85, method=6)
    return rel


def main():
    if os.path.isdir(f'{OUT}/images'):
        shutil.rmtree(f'{OUT}/images')
    os.makedirs(OUT, exist_ok=True)
    research = {}
    for group in ('launch', 'unite'):
        path = f'{SP}/research/{group}/results.json'
        if os.path.exists(path):
            for row in json.load(open(path)):
                research[row['key']] = row
    products, report = [], []
    for key, (slug, category, brands, page, title) in CATALOG.items():
        row = research.get(key)
        o = OVERRIDES.get(key, {})
        if o.get('skip'):
            report.append((slug, 'SKIPPED', o['skip'], 0, None))
            continue
        if o.get('portfolio_only'):
            row = None
        if 'brands' in o:
            brands = o['brands']
        if row and row.get('found') and row.get('summary'):
            sources = [i['file'] for i in row.get('images') or [] if os.path.exists(i['file'])]
            if page and (o.get('pdf_image_first') or not sources):
                sources = [PDF_IMG.format(page)] + sources
            elif page and o.get('pdf_image_second'):
                sources = sources[:1] + [PDF_IMG.format(page)] + sources[1:]
            images = [save_image(f, slug, n + 1) for n, f in enumerate(sources[:4])]
            specs = [t for t in row.get('technicalTable') or [] if t.get('label') not in o.get('drop_specs', set())]
            products.append(dict(slug=slug, name=o.get('name', row['name']), category=category, brands=brands, summary=row['summary'],
                                 description=row.get('description') or row['summary'], features=row.get('features') or [],
                                 functions=row.get('functions') or [], technicalTable=specs,
                                 images=images, source=row.get('source_url'), portfolio_title=title))
            report.append((slug, 'official', row.get('source_url'), len(images), row.get('match_confidence')))
        elif page and key in PDF_FALLBACK_SUMMARY:
            products.append(dict(slug=slug, name=title, category=category, brands=brands, summary=PDF_FALLBACK_SUMMARY[key],
                                 images=[save_image(PDF_IMG.format(page), slug, 1)], source=f'portfolio PDF p{page}', portfolio_title=title))
            report.append((slug, 'portfolio', f'p{page}', 1, None))
        else:
            report.append((slug, 'MISSING', row.get('notes') if row else 'no research result', 0, None))
    for p in PDF_ONLY:
        products.append(dict(slug=p['slug'], name=p['name'], category=p['category'], brands=p['brands'], summary=p['summary'],
                             images=[save_image(PDF_IMG.format(p['page']), p['slug'], 1)], source=f"portfolio PDF p{p['page']}",
                             portfolio_title=p['name']))
        report.append((p['slug'], 'portfolio', f"p{p['page']}", 1, None))
    json.dump(products, open(f'{OUT}/products.json', 'w'), indent=1, ensure_ascii=False)
    for r in report:
        print(*r, sep=' | ')
    print(len(products), 'products written')


main()
